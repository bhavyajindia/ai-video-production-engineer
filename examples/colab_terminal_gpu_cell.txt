# Option 2: Use ttyd for web terminal (easier, no ngrok needed)
!pip install -q pyngrok
from pyngrok import ngrok
!wget -q https://github.com/tsl0922/ttyd/releases/download/1.7.4/ttyd.x86_64 -O /usr/local/bin/ttyd && chmod +x /usr/local/bin/ttyd
!ttyd -p 7681 bash &
public_url = ngrok.connect(7681, "http")
print("Web Terminal URL:", public_url)
